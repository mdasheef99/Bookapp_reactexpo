function deferred() {
    let resolve!: () => void; let reject!: (error: Error) => void;
    const promise = new Promise<void>((yes, no) => { resolve = yes; reject = no; });
    return { promise, resolve, reject };
}
async function screen() {
    const result = render(<QueryClientProvider client={client}><BookDetailScreen /></QueryClientProvider>);
    await result.findByText('Fixture book');
    return result;
}
async function reviewScreen() {
    const result = await screen();
    const entry = result.queryByRole('button', { name: 'My review' });
    if (entry) fireEvent.press(entry);
    await result.findByTestId('library-review-input');
    return result;
}
beforeEach(() => {
    jest.clearAllMocks();
    mockEntryId = 'entry';
    mockActorId = 'reader';
    client = new QueryClient({ defaultOptions: { queries: { retry: false, gcTime: Infinity }, mutations: { retry: false, gcTime: Infinity } } });
    mockBook = { id: 'entry', book_id: 'catalogue', review: 'Confirmed review', review_is_public: true, rating: null,
        ownership: 'owned', reading_status: 'reading', condition: 'good', available_for_lending: false,
        book: { id: 'catalogue', title: 'Fixture book', authors: ['Author'], publisher: 'Press', page_count: 320, isbn_13: '123' } };
    (booksService.getBookDetails as jest.Mock).mockImplementation(async () => mockBook);
    (booksService.getPublicReviewsForBook as jest.Mock).mockResolvedValue([]);
    (manualNotesGateway.list as jest.Mock).mockResolvedValue([{ id: 'note', content: 'Saved note preview' }]);
    (booksService.addRating as jest.Mock).mockImplementation(async (_id, rating, review, visibility) => { mockBook = { ...mockBook, rating: rating ?? mockBook.rating, review, review_is_public: visibility }; });
    jest.spyOn(Alert, 'alert').mockImplementation(() => {});
});
afterEach(() => { cleanup(); client.clear(); jest.restoreAllMocks(); });

it('reproduces an old actor queued write dispatching after account change', async () => {
    const request = deferred();
    const observedActors: string[] = [];
    (booksService.addRating as jest.Mock).mockImplementation(() => {
        observedActors.push(mockActorId);
        return observedActors.length === 1 ? request.promise : Promise.resolve();
    });
    const ui = await reviewScreen();
    fireEvent(ui.getByTestId('library-review-input'), 'blur');
    await waitFor(() => expect(booksService.addRating).toHaveBeenCalledTimes(1));
    fireEvent.changeText(ui.getByTestId('library-review-input'), 'Actor A private review');
    fireEvent.press(ui.getByTestId('library-review-privacy-toggle'));
    ui.unmount();
    mockActorId = 'other-reader';
    await act(async () => request.resolve());
    await waitFor(() => expect(booksService.addRating).toHaveBeenCalledTimes(2));
    expect(observedActors).toEqual(['reader', 'other-reader']);
    expect(booksService.addRating).toHaveBeenNthCalledWith(2, 'entry', undefined, 'Actor A private review', false);
    await waitFor(() => expect(client.isMutating()).toBe(0));
    expect(client.getQueryData<any>(['book', 'reader', 'entry'])).toMatchObject({ review: 'Actor A private review', review_is_public: false });
});

it('reproduces newer typing back to the confirmed value being overwritten on response', async () => {
    const request = deferred();
    (booksService.addRating as jest.Mock).mockReturnValue(request.promise);
    const ui = await reviewScreen();
    fireEvent.changeText(ui.getByTestId('library-review-input'), 'Submitted B');
    fireEvent(ui.getByTestId('library-review-input'), 'blur');
    await waitFor(() => expect(booksService.addRating).toHaveBeenCalledTimes(1));
    fireEvent.changeText(ui.getByTestId('library-review-input'), 'Confirmed review');
    await act(async () => { mockBook = { ...mockBook, review: 'Submitted B' }; request.resolve(); });
    await waitFor(() => expect(client.isMutating()).toBe(0));
    await waitFor(() => expect(ui.getByTestId('library-review-input').props.value).toBe('Submitted B'));
});

it('reproduces missing busy feedback for inherited pending writes after reopening', async () => {
    const request = deferred();
    (booksService.addRating as jest.Mock).mockReturnValue(request.promise);
    const initial = await reviewScreen();
    fireEvent.press(initial.getByTestId('library-review-privacy-toggle'));
    await waitFor(() => expect(booksService.addRating).toHaveBeenCalledTimes(1));
    initial.unmount();
    const reopened = await reviewScreen();
    expect(client.isMutating({ mutationKey: ['book-review', 'reader', 'entry'], exact: true })).toBe(1);
    expect(reopened.getByTestId('library-review-privacy-toggle').props.accessibilityState.busy).toBe(false);
    expect(reopened.queryByText(/Saving review changes/)).toBeNull();
    await act(async () => request.reject(new Error('Synthetic stop')));
    await waitFor(() => expect(client.isMutating()).toBe(0));
});
